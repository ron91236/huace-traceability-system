# 长图溯源模板（企业自定义模块自由拼接）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让企业自助配置若干"长图模块"（上传 PDF/长图→系统转整页图→纵向无缝拼接），消费者扫码页以整屏替换 + 底部胶囊导航的方式展示，检测报告在放大后可左右切份、张内上下滚动。

**Architecture:** 新增 `trace_template` 行（`template_key='long-image'`，`config_json.renderMode='long-image'`）作为纯样式开关，权限沿用现有 `enterprise.assigned_template_ids` 分配链路；企业内容落在新表 `trace_long_module`（企业默认 + 商品级覆盖）；消费端 `TracePageService` 仅在 `renderMode=long-image` 时附带 `longModules`，前端新增 `LongImageBoard.vue` 承接整屏渲染，`TraceView.vue` 只做数据所有者与分支。

**Tech Stack:** Spring Boot 3.2.5 + MyBatis-Plus + PDFBox + Redis；Vue 3 `<script setup>` + TypeScript + Element Plus 2.7 + sortablejs + Vite。

**Spec:** `docs/superpowers/specs/2026-10-22-长图溯源模板-design.md`

```sql
-- 长图溯源模板：企业自定义模块（整页图自由拼接）+ long-image 模板行
SET @tbl_exists = (SELECT COUNT(*) FROM information_schema.TABLES
    WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'trace_long_module');
SET @ddl = IF(@tbl_exists = 0,
    'CREATE TABLE trace_long_module (
        id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT ''主键'',
        enterprise_id BIGINT NOT NULL COMMENT ''归属企业'',
        goods_id BIGINT NULL COMMENT ''归属商品，NULL=企业默认全部商品'',
        name VARCHAR(50) NOT NULL COMMENT ''模块名称，即底部导航文案'',
        pages TEXT NOT NULL COMMENT ''整页图片URL的JSON数组，顺序即拼接顺序'',
        sort_order INT NOT NULL DEFAULT 0 COMMENT ''导航顺序，小者在前'',
        status TINYINT NOT NULL DEFAULT 1 COMMENT ''1启用 0停用'',
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT ''创建时间'',
        updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT ''更新时间'',
        INDEX idx_ent_goods (enterprise_id, goods_id, status, sort_order)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT=''长图溯源模块：企业自助配置的整屏长图内容''',
    'SELECT ''trace_long_module 已存在，跳过''');
PREPARE stmt FROM @ddl; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- 模板行：只承载样式与渲染模式，不含企业内容；template_key 唯一，重复执行不报错
INSERT INTO trace_template (template_key, template_name, template_type, config_json, status)
SELECT 'long-image', '长图溯源', 'custom', '{"renderMode":"long-image"}', 1
WHERE NOT EXISTS (SELECT 1 FROM trace_template WHERE template_key = 'long-image');
```

> `trace_template.template_type` 在 `sql/V1__init_schema.sql:313-325` 是可空 `VARCHAR(20)`、`config_json` 是 `JSON` 类型、`template_key` 有 `UNIQUE KEY uk_template_key`，所以上面的插入语句无需额外约束处理。
>
> 评审后确定的三件事（已核对生产库 `information_schema`，2026-10-22）：
> 1. 生产 `trace_template` 除 `id`/`template_key` 外全部可空（`background_image` 是库外手工加的列，`IS_NULLABLE=YES`），所以 INSERT 不带它不会触发 1364。
> 2. 建表必须带 `COLLATE=utf8mb4_unicode_ci`：库默认就是它，V16/V17 也都显式指定；不写会落到 MySQL8 的 `utf8mb4_0900_ai_ci`，将来按 `name`/`pages` 跨表 JOIN/UNION 会报 Illegal mix of collations。
> 3. 整段脚本必须在**同一个连接**里跑完（`@ddl` 是会话变量），且要带 `--default-character-set=utf8mb4`，否则中文 COMMENT 与 `template_name='长图溯源'` 会存成乱码。Task 13 的命令已按此写。

> 实体上**不加** `@TableField(updateStrategy = FieldStrategy.ALWAYS)`。清空适用商品要把 NULL 写回库，这件事在 Task 4 的 `update()` 里用 `LambdaUpdateWrapper.set(TraceLongModule::getGoodsId, null)` 显式表达——这是仓库既有写法（`BatchService.java:130-133`、`AdminController.java:1004-1012`），也是全仓库目前唯一这么处理可空列的地方。把 ALWAYS 标在 POJO 上会让**将来任何** `updateById` 补丁都带上 `goods_id`，漏回填一次就把商品级模块静默降成企业默认（接口照样回 200）；放在 wrapper 里则只有真正要改归属的那条 SQL 会碰这一列。`updated_at` 由 DDL 的 `ON UPDATE CURRENT_TIMESTAMP` 兜底，`update(null, wrapper)` 不触发 `FieldFill` 也不会留旧值。

    public List<Map<String, Object>> resolveForTrace(Long enterpriseId, Long goodsId) {

        List<TraceLongModule> modules = goodsId == null ? List.of()
                : moduleMapper.selectList(new LambdaQueryWrapper<TraceLongModule>()
                        .eq(TraceLongModule::getEnterpriseId, enterpriseId)
                        .eq(TraceLongModule::getGoodsId, goodsId)
                        .eq(TraceLongModule::getStatus, 1)
                        .orderByAsc(TraceLongModule::getSortOrder));
        if (modules.isEmpty()) {
            modules = moduleMapper.selectList(new LambdaQueryWrapper<TraceLongModule>()
                    .eq(TraceLongModule::getEnterpriseId, enterpriseId)
                    .isNull(TraceLongModule::getGoodsId)
                    .eq(TraceLongModule::getStatus, 1)
                    .orderByAsc(TraceLongModule::getSortOrder));
        }
        List<Map<String, Object>> result = new ArrayList<>();
        for (TraceLongModule m : modules) {
            List<String> pages = parsePages(m.getPages());
            if (pages.isEmpty()) continue;

            item.put("key", String.valueOf(m.getId()));
            item.put("name", m.getName());
            item.put("pages", pages);
            result.add(item);
        }
        return result;
    }

                .eq(TraceLongModule::getEnterpriseId, enterpriseId)
                .orderByAsc(TraceLongModule::getSortOrder)
                .orderByAsc(TraceLongModule::getId));

    private List<String> parsePages(String json) {
        if (json == null || json.isBlank()) return List.of();

            return List.of();

    /**
     * 将PDF文件转换为图片URL列表
     * @param pdfUrl PDF文件的访问URL（可以是http URL或本地文件路径）
     * @param maxPages 页数上限，0 表示不限制；超限抛业务异常而不是静默返回空
     */
    public List<String> convertPdfToImages(String pdfUrl, int maxPages) {
        List<String> imageUrls = new ArrayList<>();
        if (pdfUrl == null || pdfUrl.isEmpty()) return imageUrls;

        try {
            byte[] pdfBytes = loadPdfBytes(pdfUrl);
            if (pdfBytes == null || pdfBytes.length == 0) {
                log.warn("PDF文件为空或无法读取: {}", pdfUrl);
                return imageUrls;
            }
            try (PDDocument document = Loader.loadPDF(pdfBytes)) {
                int totalPages = document.getNumberOfPages();
                if (maxPages > 0 && totalPages > maxPages) {
                    throw new BusinessException("PDF 共 " + totalPages + " 页，超过 " + maxPages + " 页上限，请拆分后上传");
                }
                String key = sha256Hex(pdfBytes).substring(0, 16);
                Path cacheDir = Path.of(uploadDir, "pdf-images", key);
                Files.createDirectories(cacheDir);

                List<String> cached = readCachedPages(cacheDir, key);
                if (!cached.isEmpty()) {
                    log.info("PDF转图片命中缓存: {} 页, 源: {}", cached.size(), pdfUrl);
                    return cached;
                }

                PDFRenderer renderer = new PDFRenderer(document);
                for (int i = 0; i < totalPages; i++) {
                    BufferedImage image = renderer.renderImageWithDPI(i, RENDER_DPI, ImageType.RGB);
                    String fileName = key + "_p" + (i + 1) + ".jpg";
                    writeJpeg(image, cacheDir.resolve(fileName).toFile());
                    imageUrls.add(urlPrefix + "/pdf-images/" + key + "/" + fileName);
                }
                Files.writeString(cacheDir.resolve("done"), String(totalPages));
                log.info("PDF转图片完成: {} 页, 源: {}", totalPages, pdfUrl);
            }
        } catch (BusinessException e) {
            throw e;
        } catch (Exception e) {
            log.error("PDF转图片失败: {}", pdfUrl, e);
        }
        return imageUrls;
    }

计划原稿只给企业门加了限制，评审指出 `PdfConvertService` 本身还挂着另一扇门，于是本任务实际交付比设计稿大。定稿要点：

**两扇门、两种失败契约（不可混为一谈）**
- 企业门 `LongModuleService.convertPdf(String url)`：调用方传正数 `maxPages = MAX_PAGES(30)`，任何失败都抛 `BusinessException` 给前端 toast。

- 公开门原来的"不限页数"（`maxPages=0`）改成 `HARD_PAGE_LIMIT = 100`；传 0 仍按 100 兜底。

**目录收口（评审的 Critical，属计划分层错误而非实现错误）**
`loadPdfBytes` 改造前会抓任意 `http(s)` URL（SSRF 到内网/元数据）、用不带锚点的 `replace("/uploads/", ...)` 拼路径、并把裸输入当磁盘路径 `new File(url)` 试。这三条挂在免登录接口上，红跑日志里能看到它真的读过 `/etc/passwd`。定稿只认 `/uploads/` 前缀，两段式收口：先 `normalize()` + `startsWith(root)` 挡 `..`，再 `toRealPath()` 解符号链接后复查 `startsWith(root)`，且 `root` 自身也 `toRealPath()`（`uploadDir` 本身是链接时比较才对齐）。`LongModuleService.resolveUploadFile` 是同一段逻辑的第二次落地（两道门各自校验各自输入，可信方不同），注释里写明必须同步 —— 已知代价是六行重复。

**限量与合法性判定放在读取/解析之前**
- `Files.size(real) > MAX_PDF_BYTES(20MB)` 在 `readAllBytes` 之前判，不是读进堆之后再判；`MAX_PDF_BYTES` 提到 `PdfConvertService` 上做 `public static final`，`LongModuleService` 那份私有常量删掉，只剩一个数。
- `%PDF` 魔数在 sha256/`loadPDF` 之前判，垃圾文件不会建出缓存目录。
- `Files.createDirectories(cacheDir)` 移到页数放行之后，被拒的 PDF 不在缓存目录里留空壳。

**缓存一致性（页数判定早于 loadPDF 的前提）**
`readCachedPages` 现在把排序后的 jpg 清单条数与 `done` 里记录的整数比对，不一致或 `done` 不可解析就整体返回空，交给调用方重新解析并按真实页数限量、再回写缓存。没有这个校验，"缓存清单就是页数"只是注释里的信仰：多出的 `_pN.jpg` 会伪造页数把合法 PDF 拒掉并发出不存在的 URL，删掉的 jpg 会虚报页数绕过上限后让前端 404。

`resolveUploadFile` 与 `confineToUploads` 都接 `IOException | IllegalArgumentException`：`root.resolve(...)` 对含 NUL 的路径抛 `InvalidPathException`，它继承 `IllegalArgumentException` 而非 `IOException`，漏接会变成 500 并把用户输入回显进 `Result.error("系统内部错误: ...")`。企业门额外 `log.warn` 一次，避免 `file.upload-dir` 配置坏掉时只报"请重新上传"、运维查不到线索。

  active.value = { ...m }
  pageList.value = parsePages(m.pages)

function parsePages(json: string): string[] {
  try { const v = JSON.parse(json || '[]'); return Array.isArray(v) ? v.filter(Boolean) : [] } catch (e) { return [] }

    const up = await uploadFile(file.raw)
    const url = up.data?.url
    if (!url) return ElMessage.error('上传失败')

    ElMessage.success(`已转换 ${pages.length} 页`)

function removePage(i: number) {

function handleRevert() { select(modules.value.find(m => m.id === activeId.value)) }

    ElMessage.success('保存成功')
    await loadData()
    select(modules.value.find(m => m.name === payload.name) || modules.value[0])
  } finally { saving.value = false }
}

async function handleDelete() {
  if (!active.value?.id) { active.value = null; return }

  activeId.value = null
  await loadData()

}

          modules.value.splice(evt.newIndex, 0, moved)
          await sortLongModules(modules.value.map(m => m.id))

          const moved = pageList.value.splice(evt.oldIndex, 1)[0]
          pageList.value.splice(evt.newIndex, 0, moved)
        },
      })
    }
  })
}

onUnmounted(() => { listSortable?.destroy(); pagesSortable?.destroy() })

const reportDetailVisible = ref(false)
const currentReport = ref<any>(null)
const currentReportImages = ref<string[]>([])
const reportImagesLoading = ref(false)
const reportPageCache = ref<Record<number, string[]>>({})
const flatReportImages = ref<string[]>([])
const flatReportStart = ref<Record<number, number>>({})

function reportPagesOf(report: any): string[] {
  if (report.reportImages) {
    return Array.isArray(report.reportImages)
      ? report.reportImages
      : String(report.reportImages).split(',').filter(Boolean)
  }
  return []
}

async function loadReportPages(report: any): Promise<string[]> {
  const cached = reportPageCache.value[report.id]
  if (cached) return cached
  let pages = reportPagesOf(report)
  if (!pages.length && report.reportPdf) {
    try {
      const res = await convertPdfToImages(report.reportPdf)
      pages = res.data || []
    } catch (e) {
      console.warn('PDF转图片失败，回退iframe模式')
      pages = []
    }
  }
  reportPageCache.value = { ...reportPageCache.value, [report.id]: pages }
  return pages
}

async function openReportDetail(report: any) {
  currentReport.value = report
  reportDetailVisible.value = true
  reportImagesLoading.value = true
  try {
    const list = testReports.value
    const groups = await Promise.all(list.map(loadReportPages))
    const flat: string[] = []
    const start: Record<number, number> = {}
    list.forEach((r: any, i: number) => {
      start[r.id] = flat.length
      flat.push(...groups[i])
    })
    flatReportImages.value = flat
    flatReportStart.value = start
    currentReportImages.value = reportPageCache.value[report.id] || []
  } finally {
    reportImagesLoading.value = false
  }
}
